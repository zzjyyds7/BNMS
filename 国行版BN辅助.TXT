local function GetIcon(IconName)

local gethui = nil
pcall(function()
	gethui = gethui or (getgenv and getgenv().gethui)
	if not gethui then
		gethui = getrenv and getrenv().gethui
	end
end)

local function safeGetUI()
	local success, ui = pcall(function()
		if gethui and type(gethui) == "function" then
			return gethui()
		end
		return nil
	end)
	if success and ui then
		return ui
	end
	if syn and syn.protect_gui then
		return game:GetService("CoreGui")
	end
	return game:GetService("CoreGui")
end

function OrionLib:IsRunning()
	local parent = Orion.Parent
	return parent and parent:IsDescendantOf(game)
end

local function AddConnection(Signal, Function)
	if not OrionLib:IsRunning() then
		return
	end
	local SignalConnect = Signal:Connect(Function)
	table.insert(OrionLib.Connections, SignalConnect)
	return SignalConnect
end

task.spawn(function()
	while OrionLib:IsRunning() do
		task.wait()
	end
	for _, Connection in pairs(OrionLib.Connections) do
		pcall(function()
			Connection:Disconnect()
		end)
	end
end)

local function MakeDraggable(DragPoint, Main)
	pcall(function()
		local Dragging, DragInput, MousePos, FramePos = false
		AddConnection(DragPoint.InputBegan, function(Input)
			if Input.UserInputType == Enum.UserInputType.MouseButton1 then
				Dragging = true
				MousePos = Input.Position
				FramePos = Main.Position
				Input.Changed:Connect(function()
					if Input.UserInputState == Enum.UserInputState.End then
						Dragging = false
					end
				end)
			end
		end)
		AddConnection(DragPoint.InputChanged, function(Input)
			if Input.UserInputType == Enum.UserInputType.MouseMovement then
				DragInput = Input
			end
		end)
		AddConnection(UserInputService.InputChanged, function(Input)
			if Input == DragInput and Dragging then
				local Delta = Input.Position - MousePos
				Main.Position = UDim2.new(FramePos.X.Scale, FramePos.X.Offset + Delta.X, FramePos.Y.Scale, FramePos.Y.Offset + Delta.Y)
			end
		end)
	end)
end

local function Create(Name, Properties, Children)
	local Object = Instance.new(Name)
	for i, v in next, Properties or {} do
		Object[i] = v
	end
	for i, v in next, Children or {} do
		v.Parent = Object
	end
	return Object
end

local function CreateElement(ElementName, ElementFunction)
	OrionLib.Elements[ElementName] = function(...)
		return ElementFunction(...)
	end
end

local function MakeElement(ElementName, ...)
	return OrionLib.Elements[ElementName](...)
end

local function SetProps(Element, Props)
	for Property, Value in pairs(Props) do
		Element[Property] = Value
	end
	return Element
end

local function SetChildren(Element, Children)
	for _, Child in pairs(Children) do
		Child.Parent = Element
	end
	return Element
end

local function Round(Number, Factor)
	local Result = math.floor(Number / Factor + (math.sign(Number) * 0.5)) * Factor
	if Result < 0 then
		Result = Result + Factor
	end
	return Result
end

local function ReturnProperty(Object)
	if Object:IsA("Frame") or Object:IsA("TextButton") then
		return "BackgroundColor3"
	end
	if Object:IsA("ScrollingFrame") then
		return "ScrollBarImageColor3"
	end
	if Object:IsA("UIStroke") then
		return "Color"
	end
	if Object:IsA("TextLabel") or Object:IsA("TextBox") then
		return "TextColor3"
	end
	if Object:IsA("ImageLabel") or Object:IsA("ImageButton") then
		return "ImageColor3"
	end
end

local function AddThemeObject(Object, Type)
	if not OrionLib.ThemeObjects[Type] then
		OrionLib.ThemeObjects[Type] = {}
	end
	table.insert(OrionLib.ThemeObjects[Type], Object)
	Object[ReturnProperty(Object)] = OrionLib.Themes[OrionLib.SelectedTheme][Type]
	return Object
end

			local packetQueue = {}
			local isProcessing = false
			local function sendPacket(remoteEvent, ...)
				local args = {...}
				table.insert(packetQueue, {
					remote = remoteEvent,
					args = args,
					time = tick()
				})
				if not isProcessing then
					isProcessing = true
					task.spawn(processQueue)
				end
			end
			local function processQueue()
				while #packetQueue > 0 do
					local pkt = table.remove(packetQueue, 1)
					local delay = math.random(CONFIG.MinDelayMS, CONFIG.MaxDelayMS) / 1000
					task.wait(delay)
					local success, err = pcall(function()
						pkt.remote:FireServer(unpack(pkt.args))
					end)
					if not success then
						table.insert(packetQueue, pkt)
						task.wait(0.5)
					end
					if #packetQueue > 10 then
						task.wait(0.1)
					end
				end
				isProcessing = false
			end
			local lastSendTime = 0
			local function onRootMoved()
				local now = tick()
				if now - lastSendTime < (1 / CONFIG.MaxPacketsPerSecond) then
					return
				end
				lastSendTime = now
			end
			RunService.Heartbeat:Connect(function(deltaTime)
				onRootMoved()
			end)
			local lastHeartbeat = tick()
			local isPaused = false
			local currentVelocity = nil
			RunService.Heartbeat:Connect(function()
				local now = tick()
				local elapsed = now - lastHeartbeat
				if elapsed > CONFIG.HeartbeatTimeout then
					if not isPaused then
						isPaused = true
						if currentVelocity then
							currentVelocity:Destroy()
							currentVelocity = nil
						end
					end
				else
					if isPaused then
						isPaused = false
					end
				end
				lastHeartbeat = now
			end)

						child.OnClientEvent = function(...)
							lastHeartbeat = tick()
							if oldFunc then
								oldFunc(...)
							end
						end
					end
					if child:IsA("Folder") or child:IsA("Model") then
						hookAllRemotes(child)
					end
				end
			end

			end)
			local function hookRemoteEvents(container)
				for _, obj in ipairs(container:GetDescendants()) do
					if obj:IsA("RemoteEvent") then
						local metatable = getrawmetatable(obj)
						if metatable then
							local oldIndex = metatable.__index
							metatable.__index = function(tbl, key)
								if key == "FireServer" then
									return function(self, ...)
										local args = {...}
										if AIMBOT_ENABLED and ScriptActive and not shouldSkipEvent(self) then
											args = modifyArgs(args)
										end
										sendPacket(self, unpack(args))
									end
								end
								return oldIndex(tbl, key)
							end
						end
					end
				end
			end
			pcall(function()
				hookRemoteEvents(ReplicatedStorage)
				hookRemoteEvents(player.PlayerGui)
			end)

local function shouldSkipEvent(event)
	local name = event.Name:lower()
	local skipKeywords = {
		"chat", "ping", "animation", "reload", "switch", "equip", "inventory", "ammo",
		"sync", "pose", "status", "lobby", "menu", "score", "killfeed", "announce",
		"voice", "team", "round", "leaderboard"
	}
	for _, kw in ipairs(skipKeywords) do
		if name:find(kw) then return true end
	end
	return false
end

local function shouldSkipFunction(func)
	return shouldSkipEvent(func)
end

local function modifyArgs(args)
	local bestHead = getNearestEnemyHead()
	if not bestHead then return args end

	local char = LocalPlayer.Character
	if not char then return args end
	local tool = char:FindFirstChildOfClass("Tool")
	local gunPart = nil
	if tool then
		gunPart = tool:FindFirstChild("Handle") or tool:FindFirstChild("Barrel") or tool:FindFirstChildOfClass("BasePart")
	end

	local origin = gunPart and gunPart.Position or Camera.CFrame.Position
	local direction = (bestHead.Position - origin).Unit

	local vecCount = 0
	for i, arg in ipairs(args) do
		if typeof(arg) == "Vector3" then
			vecCount = vecCount + 1
			if vecCount == 1 then
				args[i] = bestHead.Position
			elseif vecCount == 2 then
				args[i] = direction
				break
			end
		end
	end
	return args
end

local function hookRemoteFunction(remoteFunc)
	if not remoteFunc:IsA("RemoteFunction") or hookedRemoteFunctions[remoteFunc] then return end
	local oldInvoke = remoteFunc.InvokeServer
	hookedRemoteFunctions[remoteFunc] = oldInvoke
	remoteFunc.InvokeServer = function(self, ...)
		local args = {...}
		if AIMBOT_ENABLED and ScriptActive and not shouldSkipFunction(self) then
			args = modifyArgs(args)
		end
		return oldInvoke(self, table.unpack(args))
	end
end

local function unhookRemoteFunction(remoteFunc)
	if hookedRemoteFunctions[remoteFunc] then
		remoteFunc.InvokeServer = hookedRemoteFunctions[remoteFunc]
		hookedRemoteFunctions[remoteFunc] = nil
	end
end

local function hookAllRemoteFunctions()
	for _, container in ipairs(getEventContainers()) do
		for _, descendant in ipairs(container:GetDescendants()) do
			if descendant:IsA("RemoteFunction") then
				hookRemoteFunction(descendant)
			end
		end
	end
end

local function unhookAllRemoteFunctions()
	for remoteFunc, _ in pairs(hookedRemoteFunctions) do
		remoteFunc.InvokeServer = hookedRemoteFunctions[remoteFunc]
	end
	hookedRemoteFunctions = {}
end

local function startListeningForNewEvents()
	if listeningForNew then return end
	listeningForNew = true
	for _, container in ipairs(getEventContainers()) do
		container.DescendantAdded:Connect(function(descendant)
			if AIMBOT_ENABLED then
				if descendant:IsA("RemoteFunction") then
					hookRemoteFunction(descendant)
				end
			end
		end)
	end
end

local function applyRemoteHooks(enabled)
	if enabled then
		hookAllRemoteFunctions()
		startListeningForNewEvents()
	else
		unhookAllRemoteFunctions()
	end
end

local function updateCircle()
	if not fovCircle then return end
	if not ScriptActive or not AIMBOT_ENABLED or not SHOW_CIRCLE then
		fovCircle.Visible = false
		return
	end
	local bestHead = getNearestEnemyHead()
	if not bestHead then
		fovCircle.Visible = false
		return
	end
	local camPos = Camera.CFrame.Position
	local headPos = bestHead.Position
	local distance = (headPos - camPos).Magnitude
	local worldRadius = distance * math.tan(math.rad(FOV_ANGLE))

	local headScreen, onScreen = Camera:WorldToViewportPoint(headPos)
	if not onScreen then
		fovCircle.Visible = false
		return
	end
	local offsetScreen = Camera:WorldToViewportPoint(headPos + Vector3.new(worldRadius, 0, 0))
	local screenRadius = (Vector2.new(offsetScreen.X, offsetScreen.Y) - Vector2.new(headScreen.X, headScreen.Y)).Magnitude
	if screenRadius < 1 then
		fovCircle.Visible = false
		return
	end
	fovCircle.Position = Vector2.new(headScreen.X, headScreen.Y)
	fovCircle.Radius = screenRadius
	fovCircle.Color = COLORS[colorIndex]
	fovCircle.Visible = true
end

function getNearestEnemyHead()
	local best = nil
	local bestAngle = FOV_ANGLE
	local camPos = Camera.CFrame.Position
	local camLook = Camera.CFrame.LookVector
	for _, plr in ipairs(Players:GetPlayers()) do
		if plr ~= LocalPlayer and plr.Character then
			local head = plr.Character:FindFirstChild("Head")
			local humanoid = plr.Character:FindFirstChildOfClass("Humanoid")
			if head and humanoid and humanoid.Health > 0 then
				local dir = (head.Position - camPos).Unit
				local dot = camLook:Dot(dir)
				if dot > 1 then dot = 1 elseif dot < -1 then dot = -1 end
				local angle = math.deg(math.acos(dot))
				if angle <= bestAngle then
					bestAngle = angle
					best = head
				end
			end
		end
	end
	return best
end

local function getNearestEnemy()
	local nearest = nil
	local nearestDist = math.huge
	local center = Camera.ViewportSize / 2
	for _, player in ipairs(Players:GetPlayers()) do
		if player ~= LocalPlayer and player.Character then
			local char = player.Character
			local root = char:FindFirstChild("HumanoidRootPart")
			local humanoid = char:FindFirstChildOfClass("Humanoid")
			if root and humanoid and humanoid.Health > 0 then
				local screenPos, onScreen = Camera:WorldToViewportPoint(root.Position)
				if onScreen then
					local dist = (Vector2.new(screenPos.X, screenPos.Y) - center).Magnitude
					if dist < nearestDist then
						nearestDist = dist
						nearest = player
					end
				end
			end
		end
	end
	return nearest
end

local function checkCover(targetPlayer)
	local char = targetPlayer.Character
	local targetPart = char:FindFirstChild("UpperTorso") or char:FindFirstChild("Torso") or char:FindFirstChild("HumanoidRootPart")
	if not targetPart then return nil end
	local cameraPos = Camera.CFrame.Position
	local direction = (targetPart.Position - cameraPos)
	local rayLength = direction.Magnitude
	local rayUnit = direction.Unit
	local rayParams = RaycastParams.new()
	rayParams.FilterType = Enum.RaycastFilterType.Blacklist
	rayParams.FilterDescendantsInstances = {LocalPlayer.Character, targetPlayer.Character}
	local success, result = pcall(function()
		return Workspace:Raycast(cameraPos, rayUnit * rayLength, rayParams)
	end)
	if not success then
		local hit = Workspace:FindPartOnRayWithIgnoreList(Ray.new(cameraPos, rayUnit * rayLength), {LocalPlayer.Character, targetPlayer.Character})
		if hit then
			if hit:IsDescendantOf(targetPlayer.Character) then
				return false
			else
				return true
			end
		else
			return false
		end
	else
		if result then
			if result.Instance:IsDescendantOf(targetPlayer.Character) then
				return false
			else
				return true
			end
		else
			return false
		end
	end
end

local meta = getrawmetatable(game)
local old = meta.__namecall
setreadonly(meta, false)
meta.__namecall = newcclosure(function(self, ...)
	local method = getnamecallmethod()
	local args = {...}
	if not checkcaller() and self == Workspace and (method == "Raycast" or method == "Spherecast" or method == "Blockcast" or method == "FindPartOnRay" or method == "FindPartOnRayWithIgnoreList" or method == "FindPartOnRayWithWhitelist") then
		if not ScriptActive or not AIMBOT_ENABLED then
			return old(self, ...)
		end
		local origin, direction
		if method == "Raycast" then
			origin = args[1]
			direction = args[2]
		elseif method == "Spherecast" then
			origin = args[1]
			direction = args[3]
		elseif method == "Blockcast" then
			origin = args[1].Position
			direction = args[3]
		else
			local ray = args[1]
			if typeof(ray) == "Ray" then
				origin = ray.Origin
				direction = ray.Direction
			end
		end
		if origin and direction then
			local bestHead = getNearestEnemyHead()
			if bestHead then
				local hitPos = bestHead.Position
				local norm = (hitPos - origin).Unit
				local dist = (hitPos - origin).Magnitude
				if method == "Raycast" or method == "Spherecast" or method == "Blockcast" then
					return {
						Instance = bestHead,
						Position = hitPos,
						Normal = norm,
						Material = Enum.Material.Plastic,
						Distance = dist
					}
				else
					return bestHead, hitPos, norm
				end
			end
		end
	end
	return old(self, ...)
end)
setreadonly(meta, true)

local function updateESP()
	if not ESP_ENABLED or not DrawingAvailable then
		hideAllESP()
		return
	end
	local players = Players:GetPlayers()
	local activePlayers = {}
	for _, plr in ipairs(players) do
		if plr ~= LocalPlayer and plr.Character and plr.Character:FindFirstChildOfClass("Humanoid") then
			table.insert(activePlayers, plr)
		end
	end
	for plr, item in pairs(espPool) do
		if not table.find(activePlayers, plr) then
			pcall(function() if item.name then item.name:Remove() end end)
			pcall(function() if item.dist then item.dist:Remove() end end)
			pcall(function() if item.health then item.health:Remove() end end)
			if item.skeletonLines then
				for _, l in ipairs(item.skeletonLines) do pcall(function() l:Remove() end) end
			end
			espPool[plr] = nil
		end
	end
	for _, item in pairs(espPool) do
		if item.name then item.name.Visible = false end
		if item.dist then item.dist.Visible = false end
		if item.health then item.health.Visible = false end
		if item.skeletonLines then
			for _, l in ipairs(item.skeletonLines) do l.Visible = false end
		end
	end
	for _, plr in ipairs(activePlayers) do
		local char = plr.Character
		local hum = char:FindFirstChildOfClass("Humanoid")
		if hum and hum.Health > 0 then
			local item = getOrCreateESP(plr)
			if not item then continue end
			local head = char:FindFirstChild("Head")
			local root = char:FindFirstChild("HumanoidRootPart")
			if head and root then
				local screenHead = Camera:WorldToViewportPoint(head.Position)
				if screenHead.Z > 0 then
					local rootPos, rootOnScreen = Camera:WorldToViewportPoint(root.Position)
					if rootOnScreen then
						if ESP_SETTINGS.Name and item.name then
							item.name.Text = plr.Name
							item.name.Position = Vector2.new(rootPos.X, rootPos.Y - 32)
							item.name.Visible = true
						end
						if ESP_SETTINGS.Distance and item.dist then
							local dist = (Camera.CFrame.Position - root.Position).Magnitude
							item.dist.Text = string.format("%.1f M", dist)
							item.dist.Position = Vector2.new(rootPos.X, rootPos.Y - 22)
							item.dist.Visible = true
						end
						if ESP_SETTINGS.Health and item.health then
							item.health.Text = string.format("%.0f HP", hum.Health)
							item.health.Position = Vector2.new(rootPos.X, rootPos.Y - 12)
							item.health.Visible = true
						end
					else
						if item.name then item.name.Visible = false end
						if item.dist then item.dist.Visible = false end
						if item.health then item.health.Visible = false end
					end
				end
			end
			if ESP_SETTINGS.Skeleton and item.skeletonLines then
				local connections = getSkeletonConnections(hum)
				for i, conn in ipairs(connections) do
					local part1 = getPart(char, conn[1])
					local part2 = getPart(char, conn[2])
					local line = item.skeletonLines[i]
					if part1 and part2 and line then
						local pos1 = Camera:WorldToViewportPoint(part1.Position)
						local pos2 = Camera:WorldToViewportPoint(part2.Position)
						if pos1.Z > 0 or pos2.Z > 0 then
							line.From = Vector2.new(pos1.X, pos1.Y)
							line.To = Vector2.new(pos2.X, pos2.Y)
							line.Visible = true
						end
					end
				end
				for i = #connections + 1, #item.skeletonLines do
					item.skeletonLines[i].Visible = false
				end
			end
		end
	end
	for _, item in pairs(espPool) do
		if item.name then item.name.Color = Color3.fromRGB(255, 0, 0) end
		if item.dist then item.dist.Color = Color3.fromRGB(255, 0, 0) end
		if item.health then item.health.Color = Color3.fromRGB(255, 0, 0) end
		if item.skeletonLines then
			for _, l in ipairs(item.skeletonLines) do l.Color = COLORS[colorIndex] end
		end
	end
end

local function getTopScreenPos(part)
	local cf = part.CFrame
	local size = part.Size
	local half = size / 2
	local corners = {
		cf:PointToWorldSpace(Vector3.new(-half.X, half.Y, -half.Z)),
		cf:PointToWorldSpace(Vector3.new( half.X, half.Y, -half.Z)),
		cf:PointToWorldSpace(Vector3.new(-half.X, half.Y, half.Z)),
		cf:PointToWorldSpace(Vector3.new( half.X, half.Y, half.Z)),
	}
	local minX, minY = math.huge, math.huge
	local allBehind = true
	for _, p in ipairs(corners) do
		local vec, onScreen = Camera:WorldToViewportPoint(p)
		if onScreen then
			allBehind = false
		end
		if vec.X < minX then minX = vec.X end
		if vec.Y < minY then minY = vec.Y end
	end
	if allBehind then return nil end
	return Vector2.new(minX, minY)
end

local function checkEnemyView()
	local player = LocalPlayer.Character
	if not player then return "" end
	local myHead = player:FindFirstChild("Head")
	local myRoot = player:FindFirstChild("HumanoidRootPart")
	local myPos = myRoot and myRoot.Position or (myHead and myHead.Position)
	if not myPos then return "" end
	local visibleEnemies = {}
	for _, other in ipairs(Players:GetPlayers()) do
		if other ~= LocalPlayer and other.Character then
			local char = other.Character
			local head = char:FindFirstChild("Head")
			local root = char:FindFirstChild("HumanoidRootPart")
			if head and root then
				local dirToMe = (myPos - head.Position).Unit
				local lookDir = head.CFrame.LookVector
				local dot = lookDir:Dot(dirToMe)
				if dot > 0.5 then
					local rayOrigin = head.Position
					local rayDirection = (myPos - head.Position)
					local rayLength = rayDirection.Magnitude
					local rayUnit = rayDirection.Unit
					local blocked = false
					local success, result = pcall(function()
						local rayParams = RaycastParams.new()
						rayParams.FilterType = Enum.RaycastFilterType.Blacklist
						rayParams.FilterDescendantsInstances = {char, player}
						local hit = workspace:Raycast(rayOrigin, rayUnit * rayLength, rayParams)
						if hit and hit.Instance then
							blocked = true
						end
					end)
					if not success then
						local hit = workspace:FindPartOnRayWithIgnoreList(
							Ray.new(rayOrigin, rayUnit * rayLength),
							{char, player}
						)
						if hit then blocked = true end
					end
					if not blocked then
						table.insert(visibleEnemies, other.Name)
					end
				end
			end
		end
	end
	if #visibleEnemies == 0 then return "" end
	return "敌人视野内有我\n敌人: " .. table.concat(visibleEnemies, " | ")
end

local frameCount = 0
local lastTime = tick()

RunService.RenderStepped:Connect(function()
	if not ScriptActive then return end
	frameCount = frameCount + 1
	local now = tick()
	if now - lastTime >= 1 then
		FPSLabel.Text = "FPS: " .. frameCount
		frameCount = 0
		lastTime = now
	end
	if SPEED_BOOST_ENABLED then
		local char = LocalPlayer.Character
		if char then
			local hum = char:FindFirstChildOfClass("Humanoid")
			if hum then hum.WalkSpeed = 22 end
		end
	end
	if HIGH_JUMP_ENABLED then
		local char = LocalPlayer.Character
		if char then
			local hum = char:FindFirstChildOfClass("Humanoid")
			if hum then hum.JumpPower = 75 end
		end
	end
	if GRAVITY_CHANGED then
		Workspace.Gravity = 150
	end
	if WIDE_ANGLE_ENABLED then
		Camera.FieldOfView = 140
	end
	if LOCK_FPS_ENABLED then
		pcall(function() setfpscap(144) end)
	end
	if ENEMY_VIEW_WARNING then
		local text = checkEnemyView()
		EnemyViewLabel.Text = text
	else
		EnemyViewLabel.Text = ""
	end
	updateCircle()
	updateCoverDisplay()
	updateESP()
	updateExtractBotESP()
	if PLAYER_COUNT_ENABLED and ESP_ENABLED and playerCountDrawing then
		local aliveCount = 0
		for _, plr in ipairs(Players:GetPlayers()) do
			if plr ~= LocalPlayer and plr.Character and plr.Character:FindFirstChildOfClass("Humanoid") and plr.Character:FindFirstChildOfClass("Humanoid").Health > 0 then
				aliveCount = aliveCount + 1
			end
		end
		playerCountDrawing.Text = "p: " .. tostring(aliveCount)
		local screenCenter = Camera.ViewportSize / 2
		playerCountDrawing.Position = Vector2.new(screenCenter.X, screenCenter.Y - 120)
		playerCountDrawing.Visible = true
	elseif playerCountDrawing then
		playerCountDrawing.Visible = false
	end
	if ANTENNA_ENABLED and ESP_ENABLED and #antennaLinePool > 0 then
		local screenCenter = Camera.ViewportSize / 2
		local fromPos = Vector2.new(screenCenter.X, screenCenter.Y - 100)
		local index = 1
		local players = Players:GetPlayers()
		for _, plr in ipairs(players) do
			if plr ~= LocalPlayer and plr.Character then
				local head = plr.Character:FindFirstChild("Head")
				if head then
					local screenPos, onScreen = Camera:WorldToViewportPoint(head.Position)
					if screenPos.Z > 0 then
						local line = antennaLinePool[index]
						if not line then
							line = Drawing.new("Line")
							line.Thickness = 0.3
							line.Color = Color3.new(1, 1, 1)
							table.insert(antennaLinePool, line)
						end
						line.From = fromPos
						line.To = Vector2.new(screenPos.X, screenPos.Y)
						line.Visible = true
						index = index + 1
					end
				end
			end
		end
		for i = index, #antennaLinePool do
			antennaLinePool[i].Visible = false
		end
	else
		for _, line in ipairs(antennaLinePool) do
			line.Visible = false
		end
	end
	if AIMBOT_ENABLED and aimLine then
		aimLine.Color = COLORS[colorIndex]
		local bestHead = getNearestEnemyHead()
		local screenCenter = Camera.ViewportSize / 2
		if bestHead then
			local headScreenPos, onScreen = Camera:WorldToViewportPoint(bestHead.Position)
			if onScreen then
				aimLine.From = Vector2.new(screenCenter.X, screenCenter.Y)
				aimLine.To = Vector2.new(headScreenPos.X, headScreenPos.Y)
				aimLine.Visible = true
			else
				aimLine.Visible = false
			end
		else
			aimLine.Visible = false
		end
	elseif aimLine then
		aimLine.Visible = false
	end
end)
